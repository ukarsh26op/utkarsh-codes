#!/usr/bin/env bash # Utkarsh Panel - all-in-one installer # Usage: # bash <(curl -fsSL https://YOUR-SHORT-URL/) # menu # bash <(curl -fsSL https://YOUR-SHORT-URL/) all # run everything # bash <(curl -fsSL https://YOUR-SHORT-URL/) panel|nebula|nobita|jishnu|playit|repair|repair-wings # This script needs bash (zsh/sh handle `read -p` and arrays differently) if [ -z "${BASH_VERSION:-}" ]; then echo "Please run this installer with bash:" echo " bash <(curl -fsSL https://utkarsh-sh.install-sh.workers.dev)" return 1 2>/dev/null || exit 1 fi set -uo pipefail PANEL_DIR="${PANEL_DIR:-$HOME/pterodactyl}" # folder with docker-compose.yml PETRO_URL="https://raw.githubusercontent.com/ukarsh26op/utkarsh-panel-install.sh/refs/heads/main/petro-install.sh" PLAYIT_URL="https://raw.githubusercontent.com/ukarsh26op/playitt.gg/refs/heads/main/playit.gg%20install.sh" NOBITA_URL="https://ptero.nobitahost.in/" JISHNU_URL="https://codes.jishnu.site/" NEBULA_CMD="" # <- paste your Nebula install command here R=$'\033[1;31m'; G=$'\033[1;32m'; Y=$'\033[1;33m'; B=$'\033[1;34m' M=$'\033[1;35m'; C=$'\033[1;36m'; W=$'\033[1;37m'; N=$'\033[0m' say() { printf '\n%s==> %s%s\n' "$C" "$*" "$N"; } err() { printf '%s%s%s\n' "$R" "$*" "$N"; } # ---------- loading animations ---------- trap 'tput cnorm 2>/dev/null' EXIT trap 'echo; exit 130' INT if [[ "${LC_ALL:-${LC_CTYPE:-${LANG:-}}}" == *[Uu][Tt][Ff]* ]]; then FRAMES=(⠋ ⠙ ⠹ ⠸ ⠼ ⠴ ⠦ ⠧ ⠇ ⠏); FULL="█"; EMPTY="░"; OKM="✔"; BADM="✘" else FRAMES=('-' '\' '|' '/'); FULL="#"; EMPTY="-"; OKM="OK"; BADM="X" fi # spinner - animates while the process is running spinner() { local i=0 tput civis 2>/dev/null while kill -0 "$1" 2>/dev/null; do printf '\r %s%s%s %s' "$C" "${FRAMES[i++ % ${#FRAMES[@]}]}" "$N" "$2" sleep 0.1 done printf '\r\033[K' tput cnorm 2>/dev/null } # run_spin - quiet (non-interactive) commands only run_spin() { local msg="$1"; shift "$@" /tmp/utk-run.log 2>&1 & local pid=$! spinner "$pid" "$msg" wait "$pid"; local rc=$? if [ "$rc" -eq 0 ]; then printf ' %s%s%s %s\n' "$G" "$OKM" "$N" "$msg" else printf ' %s%s%s %s (log: /tmp/utk-run.log)\n' "$R" "$BADM" "$N" "$msg" fi return "$rc" } # loading_bar [delay-per-step] loading_bar() { local msg="${1:-Loading}" delay="${2:-0.03}" width=30 i f e tput civis 2>/dev/null for ((i = 0; i <= width; i++)); do printf -v f '%*s' "$i" ''; f=${f// /$FULL} printf -v e '%*s' "$((width - i))" ''; e=${e// /$EMPTY} printf '\r %s%s%s [%s%s%s%s%s] %3d%%' "$W" "$msg" "$N" "$G" "$f" "$B" "$e" "$N" "$((i * 100 / width))" sleep "$delay" done printf '\r\033[K' tput cnorm 2>/dev/null } need_panel_dir() { cd "$PANEL_DIR" 2>/dev/null || { err "Panel folder not found: $PANEL_DIR (set PANEL_DIR=...)"; return 1; } } # ---------- Repair / restart (keeps all data) ---------- # Fixes cases like: Cloudflare tunnel connected but panel shows 502 because # a container is stuck (e.g. "RWLayer of container ... is unexpectedly nil"). # Only CONTAINERS are removed - volumes (database + panel files) are kept. repair_panel() { say "Repair / start panel (keeps your data)" need_panel_dir || return 1 if ! docker compose up -d; then err "compose up failed, removing stale containers (volumes are kept) and retrying..." local proj ids proj="$(basename "$PANEL_DIR")" ids="$(docker ps -aq --filter "label=com.docker.compose.project=${proj}" 2>/dev/null || true)" [ -n "$ids" ] && docker rm -f $ids >/dev/null 2>&1 docker compose up -d || { err "Still failing. Check: cd $PANEL_DIR && docker compose logs --tail=60"; return 1; } fi # make sure the Cloudflare tunnel container is up too if docker ps -a --format '{{.Names}}' | grep -qx 'utkarsh-cloudflared'; then docker start utkarsh-cloudflared >/dev/null 2>&1 || true fi printf ' Waiting for the panel on port 80 ' local i code for i in $(seq 1 40); do code="$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:80/auth/login || true)" if [ "$code" = "200" ] || [ "$code" = "302" ]; then printf '\n %s%s%s Panel is answering (HTTP %s)\n' "$G" "$OKM" "$N" "$code" return 0 fi printf '.'; sleep 3 done printf '\n' err "Panel not answering yet. Check: cd $PANEL_DIR && docker compose logs --tail=60 panel" return 1 } # ---------- Repair / restart Wings (keeps all data) ---------- # Fixes Wings not coming back after a restart (tunnel up, wings domain 502): # stale pid, stuck/broken containers (e.g. "RWLayer ... is unexpectedly nil"), # or a missing pterodactyl_nw network. Server files live in # $WINGS_DIR/var/lib/pterodactyl/volumes, NOT in containers, so removing # stopped containers is safe - Wings recreates them when a server starts. WINGS_DIR="${WINGS_DIR:-$HOME/wings}" repair_wings() { say "Repair / restart Wings (keeps your data)" local ctl="$WINGS_DIR/wings-ctl.sh" [ -x "$ctl" ] || { err "Wings is not installed ($ctl missing). Install Wings first."; return 1; } if ! docker info >/dev/null 2>&1; then err "Docker is not running / not accessible. Start Docker first, then retry." return 1 fi "$ctl" stop >/dev/null 2>&1 || true # remove stuck / stopped containers on the wings network (data is kept) local ids ids="$(docker ps -aq --filter network=pterodactyl_nw \ --filter status=created --filter status=dead --filter status=exited 2>/dev/null || true)" if [ -n "$ids" ]; then echo " Removing stale containers..." docker rm -f $ids >/dev/null 2>&1 || true fi # make sure the wings docker network exists if ! docker network inspect pterodactyl_nw >/dev/null 2>&1; then echo " Recreating network pterodactyl_nw..." docker network create --driver bridge --subnet 172.18.0.0/16 --gateway 172.18.0.1 pterodactyl_nw >/dev/null 2>&1 \ || err "Could not create pterodactyl_nw (Wings will try itself)" fi # make sure the Cloudflare tunnel container is up too if docker ps -a --format '{{.Names}}' | grep -qx 'utkarsh-cloudflared'; then docker start utkarsh-cloudflared >/dev/null 2>&1 || true fi if ! "$ctl" start; then err "Wings failed to start. Check: $ctl logs" return 1 fi local code code="$(curl -s --max-time 5 -o /dev/null -w '%{http_code}' http://127.0.0.1:8080/api/system || true)" if [ -z "$code" ] || [ "$code" = "000" ]; then err "Wings process is up but the API on port 8080 is not answering. Check: $ctl logs" return 1 fi printf ' %s%s%s Wings API is responding (HTTP %s; 401 is normal)\n' "$G" "$OKM" "$N" "$code" } install_panel() { say "Panel + Blueprint + Cloudflare + Wings" echo -e "${Y}NOTE: --reset wipes and reinstalls the panel setup.${N}" read -rp "Continue? (yes/no): " ok [[ "$ok" =~ ^[Yy]([Ee][Ss])?$ ]] || { echo "Cancelled."; return 1; } local tmp; tmp="$(mktemp)" run_spin "Downloading panel installer" curl -fsSL "$PETRO_URL" -o "$tmp" || { rm -f "$tmp"; return 1; } bash "$tmp" --reset rm -f "$tmp" } install_nebula() { say "Nebula theme" if [ -z "$NEBULA_CMD" ]; then err "Nebula command is empty. Edit NEBULA_CMD in this script." return 1 fi need_panel_dir || return 1 bash -c "$NEBULA_CMD" } panel_script() { # $1=url $2=name need_panel_dir || return 1 run_spin "Downloading $2 script" docker compose exec -T panel sh -c "curl -fsSL '$1' -o /tmp/$2.sh" || return 1 docker compose exec panel bash "/tmp/$2.sh" } install_nobita() { say "Nobita themes & extensions"; panel_script "$NOBITA_URL" nobita; } install_jishnu() { say "Jishnu themes & extensions"; panel_script "$JISHNU_URL" jishnu; } install_playit() { say "playit.gg" run_spin "Downloading playit installer" curl -fsSL "$PLAYIT_URL" -o "$HOME/playit-install.sh" || return 1 bash "$HOME/playit-install.sh" } run_all() { install_panel || return install_nebula echo read -rp "Extensions: 1) Nobita 2) Jishnu 3) Skip [1/2/3]: " e case "$e" in 1) install_nobita ;; 2) install_jishnu ;; esac read -rp "Install playit.gg too? [y/N]: " p [[ "$p" =~ ^[Yy]$ ]] && install_playit say "Done." } banner() { printf '%s' "$C" cat <<'ART' _ _ _____ _ __ _ ____ ____ _ _ | | | ||_ _|| |/ / / \ | _ \ / ___| | | | | | | | | | | | ' / / _ \ | |_) |\___ \ | |_| | | |_| | | | | . \ / ___ \ | _ < ___) || _ | \___/ |_| |_|\_\/_/ \_\|_| \_\|____/ |_| |_| ART printf '%s\n' "$N" echo -e " ${W}Pterodactyl Panel + Wings + Cloudflare${N} ${M}|${N} ${Y}installer by Utkarsh${N}" echo -e " ${B}----------------------------------------------------${N}" echo } pause() { echo; read -rp " Press Enter to return to the menu... " _; } menu() { clear loading_bar "Loading installer" 0.02 while true; do clear banner echo -e " ${G}1)${N} Panel + Blueprint + Cloudflare + Wings" echo -e " ${G}2)${N} Nebula theme" echo -e " ${Y}3)${N} Nobita themes & extensions" echo -e " ${Y}4)${N} Jishnu themes & extensions" echo -e " ${B}5)${N} playit.gg" echo -e " ${M}6)${N} Run all ${C}(1 -> 2 -> extensions -> playit)${N}" echo -e " ${G}7)${N} Repair / restart panel ${C}(keeps data)${N}" echo -e " ${G}8)${N} Repair / restart Wings ${C}(keeps data)${N}" echo -e " ${R}0)${N} Exit" echo read -rp " Choose [0-8]: " c case "$c" in 1) loading_bar "Preparing" ; install_panel; pause ;; 2) loading_bar "Preparing" ; install_nebula; pause ;; 3) loading_bar "Preparing" ; install_nobita; pause ;; 4) loading_bar "Preparing" ; install_jishnu; pause ;; 5) loading_bar "Preparing" ; install_playit; pause ;; 6) loading_bar "Preparing" ; run_all; pause ;; 7) loading_bar "Preparing" ; repair_panel; pause ;; 8) loading_bar "Preparing" ; repair_wings; pause ;; 0) exit 0 ;; *) err "Invalid choice"; sleep 1 ;; esac done } case "${1:-menu}" in menu) menu ;; all) run_all ;; panel) install_panel ;; nebula) install_nebula ;; nobita) install_nobita ;; jishnu) install_jishnu ;; playit) install_playit ;; repair) repair_panel ;; repair-wings) repair_wings ;; *) err "Unknown option: $1"; exit 1 ;; esac
